function filterFolders(query) { const folders = document.querySelectorAll('.folder'); folders.forEach(folder => { const name = folder.querySelector('.folder-name').textContent.toLowerCase(); folder.style.display = name.includes(query.toLowerCase()) ? '' : 'none'; }); } // Attach listener after your existing scripts document.querySelector('.search input').addEventListener('input', e => { filterFolders(e.target.value); });
